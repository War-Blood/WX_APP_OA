# 测试用例模板

## 后端 API 集成测试模板

```javascript
const request = require('supertest');
const app = require('../src/app');

describe('[模块名] API', () => {
  let token;

  beforeAll(async () => {
    // 获取测试 token
    const res = await request(app)
      .post('/api/auth/login')
      .send({ code: 'test_code' });
    token = res.body.data.token;
  });

  describe('GET /api/xxx/list', () => {
    it('should return list with valid params', async () => {
      const res = await request(app)
        .get('/api/xxx/list')
        .query({ page: 1, pageSize: 20 })
        .set('Authorization', `Bearer ${token}`);

      expect(res.status).toBe(200);
      expect(res.body.code).toBe(0);
      expect(Array.isArray(res.body.data.list)).toBe(true);
    });

    it('should return 401 without token', async () => {
      const res = await request(app)
        .get('/api/xxx/list');

      expect(res.status).toBe(401);
    });
  });

  describe('POST /api/xxx/create', () => {
    it('should create successfully with valid body', async () => {
      const res = await request(app)
        .post('/api/xxx/create')
        .send({ name: 'test', type: 1 })
        .set('Authorization', `Bearer ${token}`);

      expect(res.status).toBe(200);
      expect(res.body.code).toBe(0);
    });

    it('should return 1001 with invalid body', async () => {
      const res = await request(app)
        .post('/api/xxx/create')
        .send({}) // 缺少必填字段
        .set('Authorization', `Bearer ${token}`);

      expect(res.body.code).toBe(1001);
    });
  });
});
```

## 单元测试模板

```javascript
const { someFunction } = require('../src/utils/some-module');

describe('someFunction', () => {
  it('should return correct result when input is valid', () => {
    const result = someFunction({ key: 'value' });
    expect(result).toEqual({ expected: 'output' });
  });

  it('should throw when input is invalid', () => {
    expect(() => someFunction(null)).toThrow('Invalid input');
  });

  it('should handle empty input gracefully', () => {
    const result = someFunction({});
    expect(result).toBeDefined();
  });
});
```
