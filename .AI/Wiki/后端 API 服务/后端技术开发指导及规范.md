## 1. 分层架构规范

### 1.1 架构总览

```
┌─────────────────────────────────────────────────────────┐
│                    客户端请求                             │
└──────────────────────┬──────────────────────────────────┘
                       ▼
┌─────────────────────────────────────────────────────────┐
│  全局中间件: helmet → cors → rate-limit → json → logger │
└──────────────────────┬──────────────────────────────────┘
                       ▼
┌─────────────────────────────────────────────────────────┐
│  routes/        路由层    请求路由分发、中间件绑定       │
├─────────────────────────────────────────────────────────┤
│  middleware/    中间件层  JWT 认证、参数校验、错误处理    │
├─────────────────────────────────────────────────────────┤
│  controllers/   控制器层  参数校验、请求响应封装         │
├─────────────────────────────────────────────────────────┤
│  services/      服务层    业务逻辑编排、事务管理         │
├─────────────────────────────────────────────────────────┤
│  config/        数据层    MySQL 连接池、Redis 客户端     │
└─────────────────────────────────────────────────────────┘
```

### 1.2 各层职责与约束

#### 路由层（routes/）

**职责：** 仅做 URL 路径到控制器的映射，不做任何业务处理。

```javascript
// ✅ 正确：路由层只做分发
const express = require('express');
const router = express.Router();
const healthController = require('../controllers/health');

router.get('/health', healthController.check);

module.exports = router;
```

**约束：**
- 每个模块一个路由文件，如 `health.js`、`auth.js`、`approval.js`
- 路由路径统一以模块名命名：`/api/health`、`/api/auth`、`/api/approval`
- 路由聚合统一在 `routes/index.js` 中完成
- 所有业务模块路由通过 `auth` 中间件保护（除了登录和健康检查）

#### 中间件层（middleware/）

**职责：** 请求预处理（认证、校验、日志、限流），不操作数据库。

```javascript
// ✅ 正确：中间件只做预处理
function auth(req, res, next) {
  const token = req.headers.authorization?.replace('Bearer ', '');
  if (!token) throw new AuthError('未提供认证令牌');
  const payload = jwt.verify(token, config.jwt.secret);
  req.user = payload;
  next();
}
```

**标准中间件清单：**

| 文件 | 中间件 | 说明 |
|------|--------|------|
| `auth.js` | `auth()` / `requireRole(...roles)` | JWT 认证 + 角色鉴权（M3 完善） |
| `validator.js` | `validate(schema, source)` | Joi 参数校验工厂 |
| `errorHandler.js` | `errorHandler(err, req, res, next)` | 全局错误处理 |

#### 控制器层（controllers/）

**职责：** 参数校验 + 调用服务层 + 格式化响应。不包含业务逻辑。

```javascript
// ✅ 正确：控制器只做编排，不写业务逻辑
async getProfile(req, res, next) {
  try {
    const user = await userService.getProfile(req.user.userId);
    res.json(success(user));
  } catch (err) {
    next(err);
  }
}
```

**约束：**
- 控制器方法使用 `async/await`
- 所有异常通过 `next(err)` 交给全局错误处理
- 不直接操作数据库、不调用第三方服务
- 不包含条件判断/循环等业务逻辑

#### 服务层（services/）

**职责：** 所有业务逻辑集中在此层。可跨控制器复用。

```javascript
// ✅ 正确：业务逻辑在服务层
async function submitDailyReport(userId, reportData) {
  // 1. 校验业务规则：一天只能提交一次
  const existing = await db.query(
    'SELECT id FROM daily_reports WHERE user_id = ? AND date = ?',
    [userId, reportData.date]
  );
  if (existing.length > 0) {
    throw new AppError('2001', '今日日报已提交，请勿重复提交');
  }

  // 2. 执行事务操作
  return await db.transaction(async (conn) => {
    const [result] = await conn.execute(
      'INSERT INTO daily_reports (...) VALUES (...)',
      [userId, reportData.date, ...]
    );
    return { id: result.insertId };
  });
}
```

**约束：**
- 可调用其他服务层方法（服务间调用）
- 可调用数据层（config/database.js 或 config/redis.js）
- 不直接操作 `req` / `res` 对象
- 抛出 `AppError` 或子类错误，由控制器层捕获传递

#### 数据层（config/）

**职责：** 数据源连接管理和基础查询封装。

```javascript
// config/database.js
const mysql = require('mysql2/promise');
const pool = mysql.createPool({ /* ... */ });

module.exports = {
  query(sql, params) {
    return pool.query(sql, params);
  },
  execute(sql, params) {
    return pool.execute(sql, params);
  },
  getConnection() {
    return pool.getConnection();
  },
};
```

### 1.3 调用原则总结

```
路由层       →   只做 URL 映射，不做业务
中间件层     →   只做请求预处理，不操作数据库
控制器层     →   参数校验 + 调用服务 + 响应格式化，不写判断/循环
服务层       →   业务逻辑 + 事务管理，可调用其他服务和数据层
数据层       →   SQL 执行 + 连接管理，不做业务判断
```

## 3. 编码规范与最佳实践

### 3.1 通用规范

1. **使用 `async/await`**：所有异步操作使用 `async/await`，避免回调地狱
2. **参数化查询**：所有 SQL 使用 `?` 占位符，禁止拼接字符串
3. **统一错误处理**：业务异常使用自定义 Error 类，通过 `next(err)` 传递
4. **尽早返回**：函数中验证失败尽早 `throw` 或 `return`
5. **禁止硬编码**：所有配置值通过 `.env` 或 `constants.js` 管理
6. **JSDoc 注释**：函数和方法添加 JSDoc 注释（Swagger 依赖）

### 3.2 代码示例

```javascript
// ✅ 推荐：清晰的 JSDoc 注释 + 参数校验 + 统一错误处理
/**
 * 获取用户资料
 * @param {Object} req - Express 请求对象
 * @param {Object} res - Express 响应对象
 * @param {Function} next - Express next 函数
 */
async function getProfile(req, res, next) {
  try {
    // 参数已验证（通过中间件）
    const user = await userService.getProfile(req.user.userId);
    res.json(success(user));
  } catch (err) {
    next(err);
  }
}

// ❌ 不推荐：直接在控制器中处理数据库
async function getProfile(req, res) {
  const [rows] = await db.query('SELECT * FROM users WHERE id = ?', [req.user.userId]);
  res.json({ code: 0, message: 'success', data: rows[0] });
}
```

### 3.3 错误处理最佳实践

```javascript
// ✅ 推荐：使用自定义错误类
if (!user) {
  throw new NotFoundError('用户不存在');
}

if (user.status !== 'active') {
  throw new AppError(2001, '该用户已被禁用');
}

// ❌ 不推荐：直接返回错误响应
if (!user) {
  return res.status(404).json({ code: 1002, message: '用户不存在' });

```

### 3.4 事务处理

```javascript
// ✅ 推荐：服务层事务封装
async function createApproval(data) {
  return await db.transaction(async (conn) => {
    // 1. 插入审批主表
    const [approvalResult] = await conn.execute(
      'INSERT INTO approval_instances (...) VALUES (...)',
      [data.userId, data.type, ...]
    );

    // 2. 插入审批流程节点
    await conn.execute(
      'INSERT INTO approval_flow_nodes (...) VALUES (...)',
      [approvalResult.insertId, data.approverId, ...]
    );

    // 3. 创建消息通知
    await conn.execute(
      'INSERT INTO messages (...) VALUES (...)',
      [data.approverId, 'approval', ...]
    );

    return { id: approvalResult.insertId };
  });
}
```

### 3.5 参数校验

```javascript

const Joi = require('joi');

const createApprovalSchema = Joi.object({
  type: Joi.string().valid('leave', 'expense', 'seal', 'travel', 'purchase', 'general').required(),
  title: Joi.string().min(2).max(100).required(),

// 在路由中使用
router.post('/approval/create',
```

## 4. Git 分支策略

### 4.1 分支模型

```
master (生产分支)
  └── dev (开发主分支)
        ├── feature/m2-base-arch    ← M2 基础架构
        ├── feature/m3-auth         ← M3 认证模块
        ├── feature/m4-approval     ← M4 审批模块
        ├── feature/m5-report       ← M5 日报模块
        └── fix/xxx                 ← Bug 修复分支
```

### 4.2 分支规则

| 分支 | 用途 | 说明 |
|------|------|------|
| `master` | 生产就绪代码 | 受保护，禁止直接推送 |
| `dev` | 开发主分支 | 日常开发的基础分支 |
| `feature/*` | 功能开发分支 | 从 `dev` 拉出，完成后合并回 `dev` |
| `fix/*` | Bug 修复分支 | 从 `dev` 或 `master` 拉出 |

### 4.3 开发流程

```bash
# 1. 创建并切换到 dev 分支
git checkout -b dev

# 2. 从 dev 拉取功能分支
git checkout -b feature/m2-base-arch

# 3. 开发完成后，切回 dev 合并
git checkout dev
git merge feature/m2-base-arch

# 4. 推送 dev
git push origin dev

# 5. 稳定后合并到 master
git checkout master
git merge dev
git tag v1.0.0-m2
git push origin master --tags
```

### 4.4 Commit Message 规范

```
<type>(<scope>): <description>

# type 类型
feat:    新功能
fix:     Bug 修复
docs:    文档更新
style:   代码格式（不影响逻辑）
refactor: 重构（不新增功能也不修 Bug）
test:    测试相关
chore:   构建/工具/依赖变更
perf:    性能优化

# 示例
feat(health): 实现健康检查端点
fix(auth): 修复 Token 过期后未返回 401 的问题
docs(api): 更新审批接口文档
chore(config): 添加 Redis 配置
test(utils): 添加 response.js 单元测试
```

## 5. 异常处理规范

### 5.1 自定义错误类体系

```javascript
// src/utils/errors.js
class AppError extends Error {
  constructor(code, message, data = null) {
    super(message);
    this.code = code;
    this.httpStatus = 200;  // 默认 200（业务错误）
    this.data = data;
  }
}

class ValidationError extends AppError {
  constructor(message) {
    super(1001, message);
    this.httpStatus = 400;
  }
}

class AuthError extends AppError {
  constructor(message = '未授权访问') {
    super(401, message);
    this.httpStatus = 401;
  }
}

class ForbiddenError extends AppError {
  constructor(message = '无权限访问') {
    super(403, message);
    this.httpStatus = 403;
  }
}

class NotFoundError extends AppError {
  constructor(message = '资源不存在') {
    super(1002, message);
    this.httpStatus = 404;
  }
}

module.exports = { AppError, ValidationError, AuthError, ForbiddenError, NotFoundError };
```

### 5.2 全局错误处理中间件

```javascript
// src/middleware/errorHandler.js
const logger = require('../utils/logger');
const { fail } = require('../utils/response');

function errorHandler(err, req, res, next) {
  // 1. 记录错误日志
  if (err.httpStatus === 500 || !err.httpStatus) {
    logger.error('服务器内部错误', {
      error: err.message,
      stack: err.stack,
      url: req.originalUrl,
      method: req.method,
    });
  }

  // 2. 区分自定义错误 vs 未知错误
  const httpStatus = err.httpStatus || 500;
  const code = err.code || 500;
  const message = err.httpStatus
    ? err.message
    : '服务器内部错误，请稍后重试';

  // 3. 返回错误响应
  res.status(httpStatus).json(fail(code, message, err.data || null));
}

module.exports = errorHandler;
```

## 6. 日志规范

### 6.1 日志级别

| 级别 | 用途 | 示例场景 |
|------|------|----------|
| `error` | 错误事件 | 数据库连接失败、未捕获异常、SQL 执行错误 |

### 6.3 Winston 配置示例

```javascript
// src/utils/logger.js
const winston = require('winston');
const path = require('path');
const config = require('../config/env');

const logger = winston.createLogger({
  level: config.logLevel || 'info',
  format: winston.format.combine(
    winston.format.timestamp({ format: 'YYYY-MM-DDTHH:mm:ss.SSSZ' }),
    winston.format.json()
  ),
  transports: [
    // 控制台输出
    new winston.transports.Console({
      format: winston.format.combine(
        winston.format.colorize(),
        winston.format.printf(({ timestamp, level, message, ...meta }) => {
          return `${timestamp} [${level}] ${message} ${JSON.stringify(meta)}`;
        })
      ),
    }),
    // 错误日志文件
    new winston.transports.File({
      filename: path.join(config.logDir, 'error.log'),
      level: 'error',
      maxsize: 10 * 1024 * 1024, // 10MB
      maxFiles: 10,
    }),
    // 综合日志文件
    new winston.transports.File({
      filename: path.join(config.logDir, 'combined.log'),
      maxsize: 10 * 1024 * 1024, // 10MB
      maxFiles: 30,
    }),
  ],
});

module.exports = logger;
```

## 7. 安全规范

### 7.1 必须遵守的安全规则

| 规则 | 说明 | 实施方式 |
|------|------|----------|
| 参数化查询 | 禁止 SQL 拼接 | mysql2 `?` 占位符 |
| 敏感信息加密 | 密码/密钥不入库不提交 | `.env` 管理，`.gitignore` 排除 |
| CORS 限制 | 禁止 `Access-Control-Allow-Origin: *` | 配置允许的域名白名单 |
| Helmet 安全头 | 防 XSS、点击劫持等 | `app.use(helmet())` |
| 限流保护 | 防止暴力攻击 | `express-rate-limit`：100次/15分钟 |
| 输入校验 | 所有用户输入必须校验 | Joi Schema 校验 |
| 错误信息安全 | 生产环境不暴露内部错误详情 | 全局错误处理中间件 |
| JWT 安全 | Token 签名密钥强随机 | `openssl rand -base64 32` |
| 文件上传限制 | 限制类型和大小 | M3 阶段实现 |

## 8. 数据库规范

### 8.1 建表规范

所有表必须遵循以下规范：

```sql
CREATE TABLE `table_name` (
  `id` INT UNSIGNED NOT NULL AUTO_INCREMENT COMMENT '自增主键',
  -- 业务字段...
  `status` VARCHAR(20) NOT NULL DEFAULT 'active' COMMENT '状态',
  `created_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  `updated_at` DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
  `deleted_at` DATETIME DEFAULT NULL COMMENT '软删除时间',
  PRIMARY KEY (`id`),
  KEY `idx_status` (`status`),
  KEY `idx_created_at` (`created_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='表说明';
```

**强制规则：**
- 引擎：`InnoDB`
- 字符集：`utf8mb4`
- 排序规则：`utf8mb4_unicode_ci`
- 主键：`id INT UNSIGNED AUTO_INCREMENT`
- 时间字段：`created_at`、`updated_at`、`deleted_at`（软删除）
- 所有字段必须有 `COMMENT`
- 建立必要的索引（查询频繁的字段）

### 8.2 SQL 编写规范

```javascript
// ✅ 推荐：参数化查询

  'SELECT id, user_id, nickname FROM users WHERE department = ? AND status = ?',
  [department, 'active']

// ❌ 禁止：字符串拼接

```

### 8.3 查询规范

- 禁止 `SELECT *`，始终明确指定所需字段
- 列表查询必须带分页参数
- 大数据量查询使用索引覆盖
- 复杂查询优先考虑拆分为多次简单查询

```javascript
const db = require('../config/database');
const redis = require('../config/redis');

/**
 * 执行健康检查
 * @returns {Object} 健康检查结果
 */
async function check() {
  const checks = {};

  // 检查数据库连接
  try {
    const dbStart = Date.now();
    await db.query('SELECT 1');
    checks.database = {
      status: 'ok',
      responseTime: `${Date.now() - dbStart}ms`,
    };
  } catch (err) {
    checks.database = { status: 'error', message: err.message };
  }

  // 检查 Redis 连接
  try {
    const redisStart = Date.now();
    await redis.ping();
    checks.redis = {
      status: 'ok',
      responseTime: `${Date.now() - redisStart}ms`,
    };
  } catch (err) {
    checks.redis = { status: 'error', message: err.message };
  }

  const allOk = checks.database.status === 'ok' && checks.redis.status === 'ok';

  return {
    status: allOk ? 'ok' : 'degraded',
    timestamp: new Date().toISOString(),
    uptime: process.uptime(),
    version: '1.0.0',
    checks,
  };
}

module.exports = { check };
```

### 10.4 JWT 认证中间件（桩实现）

```javascript
// src/middleware/auth.js
const jwt = require('jsonwebtoken');
const config = require('../config/env');
const { AuthError, ForbiddenError } = require('../utils/errors');

/**
 * JWT 认证中间件
 * 从 Authorization 头提取并验证 Token
 */
function auth(req, res, next) {
  const authHeader = req.headers.authorization;
  if (!authHeader || !authHeader.startsWith('Bearer ')) {
    throw new AuthError('未提供有效的认证令牌');
  }

  const token = authHeader.replace('Bearer ', '');
  try {
    const decoded = jwt.verify(token, config.jwt.secret);
    req.user = decoded;
    next();
  } catch (err) {
    if (err.name === 'TokenExpiredError') {
      throw new AuthError('Token 已过期，请重新登录');
    }
    throw new AuthError('无效的 Token');
  }
}

/**
 * 角色鉴权中间件工厂（M3 阶段完善）
 * @param  {...string} roles - 允许的角色列表
 */
function requireRole(...roles) {
  return (req, res, next) => {
    if (!req.user) throw new AuthError('未认证');
    if (!roles.includes(req.user.role)) {
      throw new ForbiddenError('无权限执行此操作');
    }
    next();
  };
}

module.exports = { auth, requireRole };
```
